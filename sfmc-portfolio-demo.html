Re-engagement audience

Find high-value customers who haven't opened an email in the last 30 days but have spent a value of greater than 500 in the last 90 days.

LEFT JOINSubquerySUM_Open Data View
Queryreengagement-1.sql
SELECT
    c.SubscriberKey,
    c.EmailAddress,
    r.TotalAmount
FROM Customers as c

/* SUBQUERY JOIN */
LEFT JOIN (
    SELECT 
        SubscriberKey, 
        SUM(Amount) as TotalAmount
    FROM Orders
    WHERE PurchaseDate >= DATEADD(day, -90, GETDATE())
    GROUP BY SubscriberKey
) as r 
ON c.SubscriberKey = r.SubscriberKey

/* KEEP YOUR OPEN JOIN HERE */
LEFT JOIN _Open as o
ON c.SubscriberKey = o.SubscriberKey
AND o.EventDate > DATEADD(day, -30, GETDATE())

WHERE c.Status = 'Active'
AND c.TotalLTV > 500
AND o.SubscriberKey is null

VIP discount logic from order history

Calculate a customer's Total Lifetime Spend based only on "Completed" orders. If they have spent over $500, show them a unique discount code.

LookupRowsFor loopTreatAsNumberFormatCurrency
ScriptDiscount-Code-A.txt
%%[

/* Declare all variables */
var @sk, @rows, @rowCount, @i, @row, @amount, @total, @status, @discount

/* Retrieve all orders by @sk */
set @sk = AttributeValue("SubscriberKey")
set @rows = LookupRows("Order_History", "SubscriberKey", @sk)
set @rowCount = RowCount(@rows)
if @rowCount > 0 then

/* Loop through and sum all completed orders placed by @sk */
set @total = 0
for @i = 1 to @rowCount do
	set @row = Row(@rows, @i)
	set @status = Field(@row, "Status")
	if @status == "Completed" then
		set @amount = TreatAsNumber(Field(@row, "Amount"))
		set @total = Add(@total, @amount)
	endif

	NEXT @i
	
endif
	
if @total >= 500 then
set @discount = "VIP20"
else
set @discount ="SAVE10"
endif

]%%

Loyalty tier messaging and order history

Create a script that greets a customer, identifies their loyalty tier from a Data Extension, and calculates how many points they need to reach the next level.

FieldFormatDateEmpty fallbackConcat
ScriptTier-Message-A.txt
%%[

/* Declare variables */
var @sk, @firstName, @rows, @rowCount, @row, @current, @tier, @msg

/* Set value for @sk */
set @sk = AttributeValue("SubscriberKey")

/* Retrieve usable data in the row of the @sk */
set @rows = Lookuprows("Loyalty_Members", "SubscriberKey", @sk)
set @rowcount = RowCount(@rows)
if @rowcount > 0 then
set @row = Row(@rows, 1)
set @tier = Field(@row, "Tier")
set @current = TreatAsNumber(Field(@row, "CurrentPoints"))
set @firstName = Field(@row, "FirstName")
endif

/* Set fallback for blank @firstName */
if Empty(@firstName) then
set @firstName = "Valued Customer"
endif

/* Calculate point difference to reach next tier & set message for each tier */
if @tier == "Bronze" then
set @diff = Subtract(500, @current)
set @msg = Concat("You need ", @diff, " more points to reach the next tier.")
elseif @tier == "Silver" then
set @diff = Subtract(1000, @current)
set @msg = Concat("You need ", @diff, " more points to reach the next tier.")
elseif @tier == "Gold" then
set @msg = "You are at the max tier."
else
set @msg = "Start shopping to earn points for rewards."
endif

]%%


Hi %%=ProperCase(@firstName)=%%, you have %%=v(@current)=%% points. %%=v(@msg)=%%.
